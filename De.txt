void Trump() {

    cout << "Unesi ostvarene bodove: ";
    cin >> ostv;
    cout << "Unesi maksimalan broj bodova: ";
    cin >> maks;

    if(ostv>maks) cout << "Računsku operaciju je nemoguće odraditi" << endl;
    double post = ostv/maks*100;

    if(post<50) ocj = 1;
    else if(post<65) ocj = 2;
    else if(post<80) ocj = 3;
    else if(post<90) ocj = 4;
    else ocj = 5;

    cout << ostv << "/" << maks << endl << post << "%" << endl << "Ocjena: " << ocj << endl;
}
